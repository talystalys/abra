#!/usr/bin/env python3
"""Measured greedy prompt lookup and RAM placement, with token-level comparisons."""
import argparse
import datetime
import json
import os
from pathlib import Path
import re
import runpy
import signal
import socket
import statistics
import subprocess
import time

ROOT = Path(__file__).resolve().parent.parent
helpers = runpy.run_path(str(ROOT/'tools/port-benchmark.py'))
DEVICES, digest = helpers['DEVICES'], helpers['digest']
p = argparse.ArgumentParser(description=__doc__)
p.add_argument('--out',type=Path,required=True)
p.add_argument('--stage',choices=['screen','quality','final','placement'],default='screen')
p.add_argument('--gpu',choices=list(DEVICES),default='secondary')
p.add_argument('--draft',type=int,default=16)
p.add_argument('--key',type=int,default=4)
p.add_argument('--configs',default='baseline,backend,lookup,lookup-backend',help='Comma-separated decoding configurations')
p.add_argument('--cases',help='Comma-separated workload IDs; default all stage workloads')
p.add_argument('--reps',type=int,help='Explicit repetitions; recorded alongside stage warmups')
p.add_argument('--complete-code',action='store_true',help='Allow 2048 generated tokens to finish code workloads; exploratory screens cap them at 512')
p.add_argument('--runtime-bin',type=Path,help='An immutable runtime snapshot for validation during a separate build')
p.add_argument('--baseline-runtime-bin', type=Path, help='Immutable libraries for the tensor-control configuration')
p.add_argument('--allreduce',choices=['none','nccl'],default='none')
a = p.parse_args()
if socket.gethostname() != 'BigBoy' or a.out.exists(): raise SystemExit('Run on BigBoy with a fresh output directory')
a.out.mkdir(parents=True)
uuid,pci,node = DEVICES[a.gpu]
inventory = subprocess.check_output(['nvidia-smi','--query-gpu=name,uuid,pci.bus_id,compute_cap,driver_version,power.limit','--format=csv,noheader'],text=True)
rows = [r for r in inventory.splitlines() if uuid in r]
if len(rows)!=1: raise SystemExit('Missing GPU')
r = [x.strip() for x in rows[0].split(',')]
if r[0]!='NVIDIA GeForce GTX 1080 Ti' or r[3]!='6.1' or r[2].lower().lstrip('0')!=pci.lower().lstrip('0'): raise SystemExit('GPU identity mismatch')
if 'tensor' in a.configs:
 for device_uuid,device_pci,_ in DEVICES.values():
  matched = [x for x in inventory.splitlines() if device_uuid in x]
  if len(matched)!=1: raise SystemExit('Missing tensor split GPU')
  fields = [x.strip() for x in matched[0].split(',')]
  if fields[0]!='NVIDIA GeForce GTX 1080 Ti' or fields[3]!='6.1' or fields[2].lower().lstrip('0')!=device_pci.lower().lstrip('0'): raise SystemExit('Tensor GPU identity mismatch')
(a.out/'hardware.csv').write_text(inventory)
source = ROOT/'.deps/llama.cpp-port'
libs = a.runtime_bin.resolve() if a.runtime_bin else source/'build-port/bin'
binary = ROOT/'build-sm61/lookup_decode'
model = ROOT/'models/Qwen3-4B-Q4_K_M.gguf'
env = {k:v for k,v in os.environ.items() if not k.startswith('ABRA_')}
env.update(CUDA_VISIBLE_DEVICES=uuid,GGML_CUDA_DISABLE_GRAPHS='1',GGML_CUDA_ALLREDUCE=a.allreduce,LD_LIBRARY_PATH=str(libs)+':/usr/local/cuda-12.9/lib64')
fixtures = json.loads((ROOT/'tools/quality-probes.json').read_text())
code = '\n'.join(f'def transform_{i}(value):\n    adjusted = value + {i}\n    return adjusted * 2\n' for i in range(24))
workloads = [
 {'id':'code-copy','prompt':'Return the following Python code exactly, without markdown or explanation:\n'+code,'limit':512},
 {'id':'code-edit','prompt':'Change each multiplication by 2 to multiplication by 3 in this code. Return only the complete code, no markdown:\n'+code,'limit':512},
 {'id':'prose','prompt':'Explain how a CPU cache works, including cache lines, locality, and why memory bandwidth matters.','limit':256},
 {'id':'reasoning','prompt':'Explain step by step how to find the sum of the integers from 1 to 100, then generalize to n.','limit':256},
]
if a.stage=='quality': workloads += [{**x,'limit':128} for x in fixtures]
if a.complete_code:
 for case in workloads:
  if case['id'] in ['code-copy','code-edit']: case['limit']=2048
if a.stage=='placement': workloads = [workloads[2]]
if a.cases:
 wanted = set(a.cases.split(',')); workloads = [x for x in workloads if x['id'] in wanted]
 if {x['id'] for x in workloads} != wanted: raise SystemExit('Unknown workload ID')
manifest = {'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'stage':a.stage,
 'complete_code':a.complete_code,
 'runtime_bin':str(libs),'allreduce':a.allreduce,
 'baseline_runtime_bin':str(a.baseline_runtime_bin.resolve()) if a.baseline_runtime_bin else None,
 'source_head':subprocess.check_output(['git','-C',str(source),'rev-parse','HEAD'],text=True).strip(),
 'model_sha256':digest(model),'binary_sha256':digest(binary),'runner_sha256':digest(Path(__file__)),
 'cpp_sha256':digest(ROOT/'tests/lookup_decode.cpp'),'libraries':{x.name:digest(x) for x in libs.glob('*.so') if not x.is_symlink()},
 'environment':{k:env[k] for k in ['CUDA_VISIBLE_DEVICES','GGML_CUDA_DISABLE_GRAPHS','GGML_CUDA_ALLREDUCE','LD_LIBRARY_PATH']},
 'clocks':'dynamic; no power/clock changes','latency_definition':'wall time of each verification block, including lookup, synchronized decoding, greedy selection and rollback; multiple tokens in a speculative block arrive together',
 'excluded_runs_per_block':5 if a.stage=='final' else 1 if a.stage=='screen' else 0,'commands':[]}
snapshot_diff = libs.parent/'source.diff'
for line in subprocess.check_output(['ldd',str(libs/'libggml-cuda.so')],text=True).splitlines():
 if 'libnccl.so' in line and '=>' in line:
  resolved = Path(line.split('=>',1)[1].strip().split()[0])
  if resolved.is_file(): manifest['nccl'] = {'path':str(resolved),'sha256':digest(resolved)}
(a.out/'source.diff').write_text(snapshot_diff.read_text() if a.runtime_bin and snapshot_diff.exists() else subprocess.check_output(['git','-C',str(source),'diff','HEAD'],text=True))
(a.out/'lookup-source.txt').write_text((ROOT/'tests/lookup_decode.cpp').read_text())
(a.out/'workloads.json').write_text(json.dumps(workloads,indent=2)+'\n')
results = {}
def percentile(values,p):
    v = sorted(values); return v[min(len(v)-1,int((len(v)-1)*p))]
for case in workloads:
 prompt = case['prompt']
 if case.get('long_context'):
    notes = ['Record '+str(i)+': the shelf contains plain paper and a small box.' for i in range(90)]
    notes.insert(43,'The access word is cobalt-orchid-731.'); prompt += '\n'+'\n'.join(notes)
 chat = '<|im_start|>user\n'+prompt+' /no_think<|im_end|>\n<|im_start|>assistant\n<think>\n\n</think>\n\n'
 prompt_file = a.out/(case['id']+'-prompt.txt'); prompt_file.write_text(chat)
 configs = [('baseline',0,{}),('backend',0,{'ABRA_TEST_BACKEND_SAMPLING':'1'}),('lookup',a.draft,{}),('lookup-backend',a.draft,{'ABRA_TEST_BACKEND_SAMPLING':'1'})]
 tensor_env = {'ABRA_TEST_TENSOR':'1','CUDA_VISIBLE_DEVICES':','.join(x[0] for x in DEVICES.values())}
 configs += [('tensor',0,tensor_env),('tensor-lookup',a.draft,tensor_env)]
 configs += [('tensor-control',0,{**tensor_env, 'LD_LIBRARY_PATH':str(a.baseline_runtime_bin.resolve())+':/usr/local/cuda-12.9/lib64' if a.baseline_runtime_bin else str(libs)+':/usr/local/cuda-12.9/lib64'}),
             ('tensor-reuse',0,{**tensor_env,'ABRA_REUSE_Q8':'1'}),
             ('tensor-output',0,{**tensor_env,'ABRA_SPLIT_TIED_OUTPUT':'1'}),
             ('tensor-output-reuse',0,{**tensor_env,'ABRA_SPLIT_TIED_OUTPUT':'1','ABRA_REUSE_Q8':'1'})]
 wanted_configs = set(a.configs.split(','))
 if a.stage!='placement':
  if not wanted_configs <= {x[0] for x in configs}: raise SystemExit('Unknown config')
  configs = [x for x in configs if x[0] in wanted_configs]
 if a.stage=='placement': configs = [('baseline',0,{}),('cpu-kv',0,{'ABRA_TEST_CPU_KV':'1'}),('layers32',0,{'ABRA_TEST_GPU_LAYERS':'32','ABRA_TEST_THREADS':'16'}),('layers24',0,{'ABRA_TEST_GPU_LAYERS':'24','ABRA_TEST_THREADS':'16'}),('q8',0,{'ABRA_TEST_CACHE_K':'q8_0','ABRA_TEST_CACHE_V':'q8_0'})]
 results[case['id']] = {}
 blocks = [configs,list(reversed(configs))] if a.stage=='final' else [configs]
 for block, order in enumerate(blocks):
  for name, draft_n, extra_env in order:
   label = f'{case["id"]}-block{block}-{name}'; prefix = a.out/label
   reps = a.reps if a.reps is not None else 15 if a.stage=='final' else 4 if a.stage=='screen' else 1
   if reps <= manifest['excluded_runs_per_block']: raise SystemExit('No measured repetitions remain')
   run_node = 0 if extra_env.get('ABRA_TEST_TENSOR')=='1' else node
   command = ['numactl',f'--cpunodebind={run_node}',f'--membind={run_node}','/usr/bin/time','-v',str(binary),str(model),str(prompt_file),str(prefix),str(case['limit']),str(draft_n),str(a.key),str(reps)]
   manifest['commands'].append({'label':label,'argv':command,'extra_environment':extra_env})
   (a.out/'manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
   print('Executing',label,flush=True)
   with (a.out/(label+'-telemetry.csv')).open('w') as tele, (a.out/(label+'-stdout.txt')).open('w') as out, (a.out/(label+'-stderr.txt')).open('w') as err:
    sampler = subprocess.Popen(['nvidia-smi','--query-gpu=timestamp,uuid,temperature.gpu,clocks.sm,memory.used,utilization.gpu','--format=csv','-lms','250'],stdout=tele,stderr=subprocess.STDOUT)
    process = subprocess.Popen(command,env={**env,**extra_env},stdout=out,stderr=err,start_new_session=True)
    started = time.monotonic()
    try:
     with (a.out/(label+'-host-memory.csv')).open('w') as memory:
      memory.write('elapsed_s,pid,VmRSS_kib,VmSwap_kib\n')
      while process.poll() is None:
       if time.monotonic()-started>1800:
        os.killpg(process.pid,signal.SIGKILL); process.wait(); raise TimeoutError('Decoding timed out')
       # numactl execs time; its direct child is the actual decoding process.
       children = Path(f'/proc/{process.pid}/task/{process.pid}/children')
       for pid in children.read_text().split() if children.exists() else []:
        try:
         status=Path(f'/proc/{pid}/status').read_text()
         if not re.search(r'^Name:\s+lookup_decode$',status,re.M): continue
         rss_value=re.search(r'^VmRSS:\s+(\d+)',status,re.M)
         swap_value=re.search(r'^VmSwap:\s+(\d+)',status,re.M)
         if rss_value: memory.write(f'{time.monotonic()-started:.3f},{pid},{rss_value[1]},{swap_value[1] if swap_value else 0}\n'); memory.flush()
        except FileNotFoundError: pass
       time.sleep(.25)
    finally:
     if process.poll() is None: os.killpg(process.pid,signal.SIGKILL); process.wait()
     sampler.terminate(); sampler.wait(timeout=10)
   if process.returncode:
    results[case['id']][label] = {'exit_code':process.returncode,'qualified':False}
    (a.out/'report.json').write_text(json.dumps(results,indent=2)+'\n')
    continue
   data = json.loads(prefix.with_suffix('.json').read_text()); runs = data['runs'][manifest['excluded_runs_per_block']:]
   tps = [1000*len(run['tokens'])/run['decode_ms'] for run in runs]
   latency = [x for run in runs for x in run['block_ms']]
   stderr = (a.out/(label+'-stderr.txt')).read_text()
   rss = re.search(r'Maximum resident set size \(kbytes\): (\d+)',stderr)
   gpu_memory=[]
   for line in (a.out/(label+'-telemetry.csv')).read_text().splitlines():
    if uuid in line: gpu_memory.append(int(line.split(',')[4].strip().split()[0]))
   record = {'samples':len(runs),'median_tps':statistics.median(tps),'min_tps':min(tps),'max_tps':max(tps),'spread_tps':max(tps)-min(tps),'p95_block_ms':percentile(latency,.95),'max_block_ms':max(latency),'tokens':runs[0]['tokens'],'repeat_tokens_identical':all(run['tokens']==runs[0]['tokens'] for run in data['runs']),'accepted':sum(run['accepted'] for run in runs),'proposed':sum(run['proposed'] for run in runs),'peak_gpu_mib':max(gpu_memory) if gpu_memory else None,'max_rss_kib':int(rss[1]) if rss else None}
   results[case['id']][label] = record
   reference = next((v for k,v in results[case['id']].items() if k.endswith('-tensor-control') and 'tokens' in v),None)
   if reference is None:
    reference = next((v for k,v in results[case['id']].items() if k.endswith('-baseline') and 'tokens' in v),None)
   if reference: record['matches_baseline_tokens']=record['tokens']==reference['tokens']
   (a.out/'report.json').write_text(json.dumps(results,indent=2)+'\n')
   print(json.dumps({k:v for k,v in record.items() if k!='tokens'}),flush=True)
